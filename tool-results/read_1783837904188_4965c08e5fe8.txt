   931→const successModal = document.getElementById('form-success');

   936→function showSuccess(title, msg) {
   937→  successTitle.textContent = title;
   938→  successMsg.textContent = msg;
   939→  successModal.style.display = 'flex';
   940→  document.body.style.overflow = 'hidden';
   941→}
   942→function hideSuccess() {
   943→  successModal.style.display = 'none';
   944→  document.body.style.overflow = '';
   945→}
   946→successClose.addEventListener('click', hideSuccess);
   947→successModal.addEventListener('click', e => { if (e.target === successModal) hideSuccess(); });
   948→document.addEventListener('keydown', e => { if (e.key === 'Escape' && successModal.style.display === 'flex') hideSuccess(); });

   950→function encode(data) {
   951→  return Object.keys(data).map(k => encodeURIComponent(k) + '=' + encodeURIComponent(data[k])).join('&');
   952→}
   953→
   954→function wireForm(form, opts) {
   955→  if (!form) return;
   956→  form.addEventListener('submit', async (e) => {
   957→    e.preventDefault();
   958→    const btn = form.querySelector('[type=submit]');
   959→    const originalLabel = btn.textContent;
   960→    btn.disabled = true;
   961→    btn.textContent = 'Sending…';
   962→
   963→    // Collect form data
   964→    const fd = new FormData(form);
   965→    const data = {};
   966→    fd.forEach((v, k) => { data[k] = v; });
   967→    // Ensure form-name is present for Netlify
   968→    if (!data['form-name']) data['form-name'] = form.getAttribute('name');
   969→
   970→    try {
   971→      const res = await fetch('/', {
   972→        method: 'POST',
   973→        headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
   974→        body: encode(data)
   975→      });
   976→      if (res.ok || res.status === 200 || res.status === 303) {
   977→        form.reset();
   978→        showSuccess(opts.successTitle, opts.successMsg);
   979→      } else {
   980→        throw new Error('Server returned ' + res.status);
   981→      }
   982→    } catch (err) {
   983→      console.error('Form submission failed:', err);
   984→      showSuccess(
   985→        'Something went wrong',
   986→        'We couldn\'t send your message right now. Please reach us directly: engriteinnovations@gmail.com or +234 813 066 5862.'
   987→      );
   988→    } finally {
   989→      btn.disabled = false;
   990→      btn.textContent = originalLabel;
   991→    }
   992→  });
   993→}
